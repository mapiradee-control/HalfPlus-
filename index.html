<script>

// โหลดค่าโควตาและประวัติเก่าจากความจำเครื่อง (ถ้ามี)

document.addEventListener("DOMContentLoaded", function() {

    if(localStorage.getItem("currentQuota")) {

        document.getElementById('maxGovDaily').value = localStorage.getItem("currentQuota");

    } else {

        document.getElementById('maxGovDaily').value = 200; // ค่าเริ่มต้นวันแรกคือ 200

    }

    updateHistoryDOM();

    calculate();

});


let currentGovShare = 0; 


function calculate() {

    let price = parseFloat(document.getElementById('price').value) || 0;

    let maxGovDaily = parseFloat(document.getElementById('maxGovDaily').value) || 0; 

    

    currentGovShare = price * 0.60;

    let userPay = 0;


    if (currentGovShare > maxGovDaily) {

        currentGovShare = maxGovDaily;

        userPay = price - maxGovDaily;

    } else {

        userPay = price * 0.40;

    }

    

    document.getElementById('govResult').innerText = currentGovShare.toLocaleString('th-TH', {minimumFractionDigits: 2, maximumFractionDigits: 2});

    document.getElementById('userResult').innerText = userPay.toLocaleString('th-TH', {minimumFractionDigits: 2, maximumFractionDigits: 2});

}


function saveTransaction() {

    let price = parseFloat(document.getElementById('price').value) || 0;

    if (price <= 0) {

        alert("กรุณาใส่ราคาสินค้าก่อนกดบันทึกครับ");

        return;

    }


    let maxGovDaily = parseFloat(document.getElementById('maxGovDaily').value) || 0;

    

    let newQuota = maxGovDaily - currentGovShare;

    if (newQuota < 0) newQuota = 0;

    

    document.getElementById('maxGovDaily').value = newQuota.toFixed(2);

    localStorage.setItem("currentQuota", newQuota.toFixed(2)); 


    let history = JSON.parse(localStorage.getItem("historyLog")) || [];

    let time = new Date().toLocaleTimeString('th-TH', {hour: '2-digit', minute:'2-digit'});

    history.push(`เวลา ${time} - ซื้อของ ${price} บ. (รัฐช่วย ${currentGovShare.toFixed(2)} บ.)`);

    localStorage.setItem("historyLog", JSON.stringify(history));


    document.getElementById('price').value = "";

    calculate();

    updateHistoryDOM();

}


function updateHistoryDOM() {

    let history = JSON.parse(localStorage.getItem("historyLog")) || [];

    let listHTML = "";

    if (history.length === 0) {

        listHTML = "<span style='color:#9ca3af; font-size:12px;'>ยังไม่มีรายการบันทึกของวันนี้</span>";

    } else {

        for(let i = history.length - 1; i >= 0; i--) { 

            listHTML += `<li class="history-item">${history[i]}</li>`;

        }

    }

    document.getElementById('historyList').innerHTML = listHTML;

}


// ปุ่มรีเซ็ตโควตากลับเป็น 200 บาท

function resetQuota() {

    if(confirm("คุณต้องการล้างประวัติและรีเซ็ตสิทธิกลับเป็น 200 บาทใช่หรือไม่?")) {

        document.getElementById('maxGovDaily').value = 200;

        localStorage.setItem("currentQuota", 200);

        localStorage.removeItem("historyLog");

        calculate();

        updateHistoryDOM();

    }

}
